// KumbhConnect — Firestore Rules
// PHASE 1 — SPARK-PLAN COMPATIBLE REDESIGN
//
// This project stays on the free Spark plan. No Cloud Functions, no Admin
// SDK backend. Every rule below is enforced ONLY by Firestore Rules and/or
// application-level (client) code. Where a field cannot be made fully

service cloud.firestore {

  match /databases/{database}/documents {

    // ================================================================
    // AUTH / ROLE HELPERS
    // ================================================================

    function signedIn() {
      return request.auth != null;
    }

    function roleDoc() {
      return get(
        /databases/$(database)/documents/mc_roles/$(request.auth.uid)
      );
    }

    function hasRole() {
      return signedIn()
        && exists(
          /databases/$(database)/documents/mc_roles/$(request.auth.uid)
        );
    }

    function isAdmin() {
      return signedIn()
        && hasRole()
        && roleDoc().data.role == 'admin'
        && roleDoc().data.active == true;
    }

    function isDept() {
      return signedIn()
        && hasRole()
        && roleDoc().data.role == 'dept'
        && roleDoc().data.active == true;
    }

    function isPress() {
      return signedIn()
        && hasRole()
        && roleDoc().data.role == 'press'
        && roleDoc().data.active == true;
    }

    // ================================================================
    // KC_LISTINGS PHOTO VALIDATION HELPERS (P0-1 hardening)
    // ----------------------------------------------------------------
    // Prior to this change, kc_listings.photos had NO server-side
    // validation at all — any signed-in owner could set it to any
    // value, of any type/size, via a direct SDK call bypassing the UI
    // entirely (confirmed in the Provider Photo Dependency Audit).
    // These helpers close that gap as tightly as Firestore Rules
    // syntax safely allows, applied identically on create and on
    // owner update.
    //
    // HONEST LIMITATION (per audit instructions — not hidden): Rules
    // cannot call out to Cloudinary to confirm an asset genuinely
    // exists at a given URL, and Firestore Rules has no generic
    // "for every item in this list" construct for arbitrary-length
    // lists — only bounded, per-index checks are possible. Because
    // the max size enforced here is exactly 3, each of the 3 possible
    // indices is checked explicitly below; this fully covers every
    // element for any list this rule allows through (size <= 3), so
    // there is no unchecked index — the only real gap is that a
    // string which merely LOOKS LIKE a valid Cloudinary URL for this
    // project's cloud_name (fzntl3lv) cannot be distinguished from an
    // asset that was actually uploaded through uploadToCloudinary().
    // Closing that fully would require a trusted backend (Cloud
    // Function on Blaze), which this project deliberately does not
    // have — see the Spark-plan note at the top of this file.
    function isValidPhotoUrl(u) {
      return u is string
        && u.size() <= 300
        && u.matches('^https://res\\.cloudinary\\.com/fzntl3lv/image/upload/[A-Za-z0-9/_.-]+$');
    }

    function isValidListingPhotos(photos) {
      return photos is list
        && photos.size() <= 3
        && (photos.size() < 1 || isValidPhotoUrl(photos[0]))
        && (photos.size() < 2 || isValidPhotoUrl(photos[1]))
        && (photos.size() < 3 || isValidPhotoUrl(photos[2]));
    }

    // Tolerates the field being entirely absent (some write paths may
    // omit it), but if present, it must be a fully valid photos list —
    // never partially/loosely checked.
    function hasValidListingPhotosField(data) {
      return !('photos' in data) || isValidListingPhotos(data.photos);
    }

    function isSelf(uid) {
      return signedIn()
        && request.auth.uid == uid;
    }

    // ================================================================
    // AUDIT: generic validators (F-05, F-06, F-09)
    // ----------------------------------------------------------------
    // NOTE: string.size() is used for every limit below. Limits are
    // generous vs. real data; the emulator test "Devanagari at limit"
    // confirms how size() counts non-Latin text before you deploy.
    // ================================================================
    function isStrMax(v, max) {
      return v is string && v.size() <= max;
    }

    function reqStrRange(v, min, max) {
      return v is string && v.size() >= min && v.size() <= max;
    }

    function optStrMax(d, f, max) {
      return !(f in d) || isStrMax(d[f], max);
    }

    function optNumberRange(d, f, lo, hi) {
      return !(f in d) || (d[f] is number && d[f] >= lo && d[f] <= hi);
    }

    // kc_users: only fields the app writes (name/email/phone/createdAt) are
    // bounded; extra keys are tolerated up to a small cap so other pages
    // that add profile fields are not broken.
    function validUserDoc(d) {
      return d.keys().size() <= 20
        && optStrMax(d, 'name', 100)
        && optStrMax(d, 'email', 254)
        && optStrMax(d, 'phone', 30);
    }

    // kc_counters: ids a client may touch. Verify against volunteer.html.
    function counterIdAllowed(id) {
      return id in ['volunteer'];
    }

    // ================================================================
    // MAP PHASE 1 — provider geo location validation
    // ----------------------------------------------------------------
    // lat/lng are OPTIONAL on kc_listings for backward compatibility
    // with listings created before this phase (they remain valid and
    // readable with no coordinates — they simply don't appear on the
    // Map). If EITHER is present, BOTH must be present, both must be
    // numeric, and both must fall within valid geographic range. This
    // never blocks a write that omits location entirely.
    function isValidLatLng(data) {
      return (!('lat' in data) && !('lng' in data))
        || (
          'lat' in data && 'lng' in data
          && (data.lat is float || data.lat is int)
          && (data.lng is float || data.lng is int)
          && data.lat >= -90 && data.lat <= 90
          && data.lng >= -180 && data.lng <= 180
        );
    }

    // ================================================================
    // AUDIT: kc_listings content validation + rating bound (F-01, F-05)
    // ================================================================
    function validListingContent(d) {
      return d.type in ['business', 'service', 'guide', 'volunteer']
        && reqStrRange(d.name, 1, 150)
        && isStrMax(d.category, 60)
        && isStrMax(d.description, 2000)
        && isStrMax(d.phone, 30)
        && isStrMax(d.address, 300)
        && isStrMax(d.servicesList, 1000);
    }

    function listingCreateKeys() {
      return [
        'ownerUid','type','name','category','description','phone','address',
        'servicesList','photos','avgRating','reviewCount','status',
        'createdAt','updatedAt','lat','lng'
      ];
    }

    // Fields an owner's EDIT may change (diff-based, so admin-added fields
    // such as approval metadata already on the doc never block an edit).
    function listingOwnerEditableKeys() {
      return [
        'type','name','category','description','phone','address',
        'servicesList','photos','status','updatedAt','lat','lng'
      ];
    }

    function reviewCreateOk(requestId, d) {
      let sr = get(/databases/$(database)/documents/kc_service_requests/$(requestId)).data;
      let ls = get(/databases/$(database)/documents/kc_listings/$(d.listingId)).data;
      return sr.uid == request.auth.uid
        && sr.status == 'completed'
        && sr.listingId == d.listingId
        && sr.get('reviewed', false) != true
        && ls.ownerUid != request.auth.uid;
    }

    // STRICT rating aggregate (F-01). The listing update must ship in the SAME
    // transaction/batch as a NEW kc_reviews document written by the caller for
    // THIS listing, and the new weighted total must grow by exactly that
    // review's rating. exists() sees the state BEFORE the write set, getAfter()
    // the state AFTER it, so a review can be "spent" only once.
    // Document accesses: exists(1) + getAfter(1) = 2 for this operation.
    function validRatingAggregateStrict(listingId, o, n) {
      let rid = n.lastReviewId;
      let rv = getAfter(/databases/$(database)/documents/kc_reviews/$(rid)).data;
      let delta = (n.avgRating * n.reviewCount) - (o.avgRating * o.reviewCount);
      return rid is string && rid.size() > 0 && rid.size() <= 100
        && !exists(/databases/$(database)/documents/kc_reviews/$(rid))
        && rv.listingId == listingId
        && rv.reviewerUid == request.auth.uid
        && rv.rating is int
        && n.avgRating is number && n.reviewCount is int
        && o.avgRating is number && o.reviewCount is int
        && n.avgRating >= 0 && n.avgRating <= 5
        && delta >= rv.rating - 0.01 && delta <= rv.rating + 0.01;
    }

    function validServiceRequest(d) {
      return d.keys().hasOnly(['uid','serviceType','listingId','providerName','name','mobile','city','date','details','status','createdAt'])
        && d.keys().hasAll(['uid','serviceType','name','mobile','status','createdAt'])
        && isStrMax(d.serviceType, 40)
        && reqStrRange(d.name, 1, 100)
        && isStrMax(d.mobile, 20)
        && optStrMax(d, 'providerName', 150)
        && optStrMax(d, 'city', 100)
        && optStrMax(d, 'date', 20)
        && optStrMax(d, 'details', 1000)
        && (!('listingId' in d) || d.listingId == null || isStrMax(d.listingId, 100))
        && d.createdAt == request.time;
    }


    // ================================================================
    // ADVERTISEMENT PHASE 2 — kc_ad_requests / kc_ads helpers
    // ----------------------------------------------------------------
    // Same Cloudinary-URL-allowlist pattern as isValidPhotoUrl above
    // (same project cloud_name, fzntl3lv) — the ad creative is uploaded
    // through the identical existing uploadToCloudinary() path used for
    // listing photos and Media Center images, not a new upload system.
    // HONEST LIMITATION: identical to the one documented on
    // isValidPhotoUrl above — a string that merely matches this
    // Cloudinary URL shape can't be distinguished from an asset that
    // was genuinely uploaded via the app's own upload call without a
    // trusted backend (Spark plan has none). Not a new gap, the same
    // one already accepted for kc_listings photos.
    function isValidAdCreativeUrl(u) {
      return u is string
        && u.size() <= 300
        && u.matches('^https://res\\.cloudinary\\.com/fzntl3lv/image/upload/[A-Za-z0-9/_.-]+$');
    }

    // Generic "optional http(s) URL" check for advertiser-supplied
    // action links (websiteUrl). Absent is fine; if present, must be
    // a plausible http(s) URL — rejects javascript:/data:/vbscript:
    // and any other unsafe scheme by construction (the regex requires
    // the string to start with http:// or https://).
    function isValidOptionalHttpUrl(data, field) {
      return !(field in data)
        || (
          data[field] is string
          && data[field].size() <= 300
          && data[field].matches('^https?://.*')
        );
    }

    // The exact set of fields an advertiser may ever touch on their own
    // kc_ad_requests document — never status, payment, or admin/approval
    // fields. Used by both CREATE (via hasOnly on the full document keys
    // being sensible) and UPDATE (via diff().affectedKeys().hasOnly()).
    function adRequestOwnerEditableKeys() {
      return [
        'businessName','contactPerson','mobile','whatsapp','email','businessCategory',
        'adTitle','adMatter','ctaText','callPhone','whatsappAction','websiteUrl','address',
        'lat','lng','placement','packageId','durationDays',
        'creativeUrl','creativeWidth','creativeHeight','creativeFormat','updatedAt'
      ];
    }

    // AUDIT (F-04): the comment above says CREATE uses a hasOnly() key
    // check; the original create rule only DENY-listed some keys. This is
    // the allowlist: editable keys + the four system keys set at creation.
    function adRequestCreateKeys() {
      return adRequestOwnerEditableKeys().concat(['uid','status','paymentStatus','createdAt']);
    }

    // Content limits mirror the form's maxlength (120/120/600) and the
    // client phone regex (<= 16 chars). Applied on CREATE and owner UPDATE
    // (the original UPDATE path re-checked only URL + lat/lng).
    function validAdRequestContent(d) {
      return reqStrRange(d.businessName, 1, 120)
        && reqStrRange(d.adTitle, 1, 120)
        && isStrMax(d.adMatter, 600)
        && optStrMax(d, 'contactPerson', 100)
        && optStrMax(d, 'mobile', 20)
        && optStrMax(d, 'whatsapp', 20)
        && optStrMax(d, 'email', 254)
        && optStrMax(d, 'businessCategory', 80)
        && optStrMax(d, 'ctaText', 60)
        && optStrMax(d, 'callPhone', 20)
        && optStrMax(d, 'whatsappAction', 20)
        && optStrMax(d, 'address', 300)
        && optStrMax(d, 'placement', 40)
        && optStrMax(d, 'packageId', 40)
        && optStrMax(d, 'creativeFormat', 20)
        && optNumberRange(d, 'durationDays', 1, 365)
        && optNumberRange(d, 'creativeWidth', 0, 20000)
        && optNumberRange(d, 'creativeHeight', 0, 20000);
    }


    // ================================================================
    // MEDIA CENTER — unchanged from previous merged version
    // ================================================================

    match /mc_photos/{id} {

      allow read: if resource.data.status == 'approved'
                  || isAdmin()
                  || (
                    isDept()
                    && resource.data.deptId == roleDoc().data.deptId
                  );

      // XSS/content-validation hardening (Media Center audit): the
      // approved-status read gate above was never the problem — it's
      // real, server-side, and already worked. The gap was that ANY
      // signed-in user could set title/caption/location/name/city/img/
      // badge to literally anything, with zero server-side constraint,
      // before an admin ever saw it. This does not replace the
      // client-side escaping fix in media-center.html — it's defense in
      // depth so a malicious payload can't even be stored to begin with,
      // regardless of how the reading page renders it.
      allow create: if signedIn()
                    && request.resource.data.section == 'user'
                    && request.resource.data.status == 'pending'
                    && request.resource.data.name is string
                    && request.resource.data.name.size() <= 100
                    && request.resource.data.city is string
                    && request.resource.data.city.size() <= 100
                    && request.resource.data.caption is string
                    && request.resource.data.caption.size() <= 300
                    && request.resource.data.location is string
                    && request.resource.data.location.size() <= 150
                    && isValidPhotoUrl(request.resource.data.img)
                    // AUDIT (F-07): the comment above lists `title` but it was never validated.
                    && optStrMax(request.resource.data, 'title', 150)
                    && (!('badge' in request.resource.data) || request.resource.data.badge == 'user');

      allow create: if isDept()
                    && request.resource.data.section == 'govt'
                    && request.resource.data.deptId == roleDoc().data.deptId
                    && request.resource.data.status == 'pending'
                    && optStrMax(request.resource.data, 'title', 200)
                    && optStrMax(request.resource.data, 'caption', 1000);

      allow create: if isAdmin();

      // AUDIT (F-07): the original combined `update, delete` clause let a
      // department flip its own `pending` documents to `approved` (the
      // create rule forces `pending`, so moderation was one update away
      // from being bypassed) and re-home documents to another dept.
      // Dept may still edit its own documents; it may not change deptId or
      // section, and may only keep the current status or send it back to
      // 'pending'. Approval stays with admin. Split because delete has no
      // request.resource.
      allow update: if isAdmin()
                    || (
                      isDept()
                      && resource.data.deptId == roleDoc().data.deptId
                      && request.resource.data.get('deptId', '') == resource.data.get('deptId', '')
                      && request.resource.data.get('section', '') == resource.data.get('section', '')
                      && (
                        request.resource.data.get('status', '') == resource.data.get('status', '')
                        || request.resource.data.get('status', '') == 'pending'
                      )
                    );

      allow delete: if isAdmin()
                    || (
                      isDept()
                      && resource.data.deptId == roleDoc().data.deptId
                    );
    }

    match /mc_press/{id} {

      allow read: if true;

      // AUDIT (F-08): `by` is the ownership key used by update/delete below,
      // but create never bound it to the caller, so a press user could
      // publish under another press member's name (and hand that member
      // control of the document).
      allow create: if isAdmin()
                    || (
                      isPress()
                      && request.resource.data.by is string
                      && request.resource.data.by == roleDoc().data.name
                    );

      // FIX (Media Center audit, Critical #2): update and delete were
      // previously combined in one clause that referenced
      // request.resource.data — which does not exist on a delete
      // operation, causing that clause to error/fail-closed and deny
      // ALL press self-deletes regardless of ownership. Split into two
      // separate clauses; delete no longer references request.resource.
      allow update: if isAdmin()
                    || (
                      isPress()
                      && resource.data.by == roleDoc().data.name
                      && request.resource.data.by == resource.data.by
                    );

      allow delete: if isAdmin()
                    || (
                      isPress()
                      && resource.data.by == roleDoc().data.name
                    );
    }

    match /mc_departments/{id} {

      allow read: if true;

      allow create, update, delete: if isAdmin();
    }

    match /mc_press_accounts/{id} {

      allow read, write: if isAdmin();
    }

    match /mc_roles/{uid} {

      allow read: if signedIn()
                  && (
                    request.auth.uid == uid
                    || isAdmin()
                  );

      allow create, update, delete: if isAdmin();
    }


    // ================================================================
    // KUMBH CONNECT USERS — unchanged
    // ================================================================

    match /kc_users/{uid} {

      allow read: if isSelf(uid) || isAdmin();

      allow create: if isSelf(uid)
                    && validUserDoc(request.resource.data);

      allow update: if isSelf(uid)
                    && validUserDoc(request.resource.data);

      // FIX (Phase 1 final audit, Critical #1): the existing
      // deleteAccountNow() self-service "Delete My Account" feature in
      // index-28.html deletes the user's own kc_users/{uid} document.
      // The prior rule only granted delete to isAdmin(), which would have
      // broken this real, existing feature. isSelf(uid) does not grant
      // deletion of ANY OTHER user's document — only the caller's own,
      // matching the same uid == request.auth.uid pattern already used
      // for read/create/update above.
      allow delete: if isSelf(uid) || isAdmin();

      allow update: if isAdmin();
    }


    // ================================================================
    // BUSINESS / SERVICE / GUIDE LISTINGS
    // ================================================================

    match /kc_listings/{id} {

      // -------------------- READ --------------------
      allow read: if resource.data.status == 'approved'
                  || isAdmin()
                  || (
                    signedIn()
                    && resource.data.ownerUid == request.auth.uid
                  );

      // -------------------- CREATE --------------------
      allow create: if signedIn()
                    && request.resource.data.ownerUid == request.auth.uid
                    && request.resource.data.status == 'pending'
                    && request.resource.data.avgRating == 0
                    && request.resource.data.reviewCount == 0
                    // P0-1: photos, if present, must be a valid list
                    // (max 3, each item a plausible Cloudinary URL for
                    // this project — see isValidListingPhotos above).
                    && hasValidListingPhotosField(request.resource.data)
                    // Map Phase 1: lat/lng, if present, must be valid
                    // numeric coordinates in range. Optional — omitting
                    // both is allowed (see isValidLatLng above).
                    && isValidLatLng(request.resource.data)
                    // AUDIT (F-05): key allowlist + content limits.
                    && request.resource.data.keys().hasOnly(listingCreateKeys())
                    && validListingContent(request.resource.data);

      // -------------------- ADMIN UPDATE (general fields) --------------------
      // CHANGED THIS REDESIGN: admin update no longer implicitly grants
      // rating-field edits. No evidence was found (media-center-admin.html
      // was never inspected) that any existing Admin feature edits
      // avgRating/reviewCount/ratingUpdatedAt directly — approval workflow
      // touches `status`, not rating fields. This is the safer default per
      // instruction #7. If Admin genuinely needs manual rating overrides,
      // that's a one-line rule change.
      // AUDIT (F-01): rating aggregate, strict mode. Placed FIRST so ordinary
      // reviewers short-circuit before the (more expensive) admin clauses.
      // REQUIRES the coordinated client change in index-94.html (submitReview
      // runs review + aggregate + reviewed flag in one transaction and writes
      // lastReviewId). Old cached clients degrade gracefully: their review is
      // saved, their aggregate write is rejected (the client already catches it).
      allow update: if signedIn()
                    && resource.data.ownerUid != request.auth.uid
                    && request.resource.data.diff(resource.data)
                       .affectedKeys()
                       .hasOnly(['avgRating', 'reviewCount', 'lastReviewId'])
                    && request.resource.data.reviewCount
                       == resource.data.reviewCount + 1
                    && validRatingAggregateStrict(id, resource.data, request.resource.data);

      allow update: if isAdmin()
                    && !request.resource.data.diff(resource.data)
                          .affectedKeys()
                          .hasAny(['avgRating', 'reviewCount', 'ratingUpdatedAt']);

      // -------------------- ADMIN UPDATE (rating fields, explicit + scoped) --------------------
      // Admin retains the ability to correct rating fields specifically —
      // e.g. to fix a bad value from client-side aggregation drift, given
      // there is no trusted backend to guarantee correctness — but ONLY
      // those fields, in a separate, narrowly-scoped rule.
      allow update: if isAdmin()
                    && request.resource.data.diff(resource.data)
                       .affectedKeys()
                       .hasOnly(['avgRating', 'reviewCount', 'ratingUpdatedAt']);

      // -------------------- OWNER EDIT --------------------
      allow update: if signedIn()
                    && resource.data.ownerUid == request.auth.uid
                    && request.resource.data.ownerUid == resource.data.ownerUid
                    && request.resource.data.status == 'pending'
                    && request.resource.data.avgRating == resource.data.avgRating
                    && request.resource.data.reviewCount == resource.data.reviewCount
                    // P0-1: identical photos validation as CREATE —
                    // applied to the resulting (post-update) value.
                    && hasValidListingPhotosField(request.resource.data)
                    // Map Phase 1: identical lat/lng validation as
                    // CREATE, applied to the resulting (post-update)
                    // value — covers both a corrected pin and a
                    // legacy listing that still has no coordinates.
                    && isValidLatLng(request.resource.data)
                    // AUDIT (F-05): owner may only touch content fields.
                    && request.resource.data.diff(resource.data).affectedKeys()
                         .hasOnly(listingOwnerEditableKeys())
                    && validListingContent(request.resource.data);

      // -------------------- RATING AGGREGATE UPDATE (client-side, Spark) --------------------
      // HONEST LIMITATION: this is the core Spark-compatible compromise.
      // No trusted backend exists to recompute avgRating/reviewCount from
      // the actual kc_reviews documents, so this rule can only bound and
      // narrow the client-side write — it CANNOT verify the new value is
      // actually correct, and CANNOT verify a real review was created.
      // What IS enforced:
      //   - only avgRating/reviewCount may change (no other field)
      //   - reviewCount may only move up by exactly 1 (no arbitrary jumps)
      //   - avgRating stays within [0,5]
      //   - the caller is NOT the listing's own owner (prevents the most
      //     obvious self-inflation case)
      // What is NOT enforced (would require a trusted backend):
      //   - that reviewCount actually equals the true number of kc_reviews
      //     documents for this listing
      //   - that avgRating was computed from real review ratings
      //   - that this specific call followed a genuine review creation at
      //     all — a signed-in non-owner could call this repeatedly via
      //     devtools/direct SDK use, incrementing reviewCount without ever
      //     creating a matching review
      // This is a deliberate, documented trade-off for staying on Spark,
      // not a hidden gap.


      // -------------------- DELETE --------------------
      allow delete: if isAdmin()
                    || (
                      signedIn()
                      && resource.data.ownerUid == request.auth.uid
                    );
    }


    // ================================================================
    // VOLUNTEERS
    // ================================================================
    // SPARK-COMPATIBLE (Phase 1 redesign): direct client creation is
    // restored (the Cloud-Function-only path required Blaze). Tightened
    // versus the ORIGINAL rule: regNo format is now validated.
    //
    // HONEST LIMITATION: Rules can require the regNo string to match the
    // expected KCV-2027-NNNNNN pattern, but cannot verify those digits
    // were actually produced by the kc_counters/volunteer increment that
    // happened moments earlier — Rules evaluate each document write
    // independently and cannot cross-reference two separate writes within
    // one client transaction. A determined attacker with direct SDK
    // access (not just using the normal UI) could still pick an arbitrary
    // 6-digit number within the pattern, potentially colliding with or
    // skipping ahead of the real sequence. This does not compromise
    // authentication or any other user's data — it only affects the
    // "niceness"/uniqueness of the registration-number sequence — but it
    // is a real, standing limitation without a trusted backend.

    match /kc_volunteers/{id} {

    // ----------------------------------------------------------------
    // R2 FINAL REMEDIATION — kc_notifications hardening
    // ----------------------------------------------------------------
    // Every notification a client may create must match EXACTLY one of
    // four typed shapes below (exact key set, read == false, createdAt ==
    // server time, bounded text). There is no generic "uid == caller"
    // create path and no blanket isAdmin() create path.
    //
    //   new_request          requester  -> listing owner
    //   request_status       provider   -> requester (completed/rejected)
    //   new_review           reviewer   -> listing owner
    //   admin_*              admin      -> any user (3 fixed types only)
    //
    // Client-created document IDs are deterministic and enforced here:
    //   req_{requestId}_new | req_{requestId}_{status} | review_{requestId}
    // A repeated .set() for the same ID resolves to an UPDATE, which the
    // update rule below denies to everyone except admin / the recipient
    // (and the recipient may only flip `read`) — duplicates safely no-op.
    //
    // HONEST LIMITATION (Spark, no backend): Rules prove the caller is a
    // genuine party to the request/review, but cannot constrain the free
    // text of title/message beyond type/size, so a genuine party can still
    // word their own (single, ID-pinned) notification arbitrarily.

    function notifKeysExactly(d, keys) {
      return d.keys().hasAll(keys) && d.keys().hasOnly(keys);
    }

    function notifBaseOk(d) {
      return d.uid is string && d.uid != ''
          && d.title is string && d.title.size() > 0 && d.title.size() <= 300
          && d.message is string && d.message.size() <= 1000
          && d.link is string
          && d.read == false
          && d.createdAt == request.time;
    }

    // ---- new_request: caller is the requester, target is the listing owner
    function newRequestNotifLinksOk(targetUid, listingId, requestId) {
      let sr = get(/databases/$(database)/documents/kc_service_requests/$(requestId)).data;
      return sr.uid == request.auth.uid
          && sr.listingId == listingId
          && targetUid == get(
               /databases/$(database)/documents/kc_listings/$(listingId)
             ).data.ownerUid;
    }

    function isValidNewRequestNotification(id, d) {
      return d.get('type', '') == 'new_request'
          && notifKeysExactly(d, ['uid','type','title','message','link','requestId','listingId','read','createdAt'])
          && notifBaseOk(d)
          && d.link == 'profile'
          && d.requestId is string && d.requestId != ''
          && d.listingId is string && d.listingId != ''
          && id == 'req_' + d.requestId + '_new'
          && exists(/databases/$(database)/documents/kc_service_requests/$(d.requestId))
          && newRequestNotifLinksOk(d.uid, d.listingId, d.requestId);
    }

    // ---- request_status: caller is the listing owner (provider), target is
    // the requester; status must be completed/rejected AND equal to the

      let sr = get(/databases/$(database)/documents/kc_service_requests/$(requestId)).data;
      return sr.status == status
          && sr.listingId == listingId
          && targetUid == sr.uid
          && get(
               /databases/$(database)/documents/kc_listings/$(listingId)
             ).data.ownerUid == request.auth.uid;
    }

    function isValidRequestStatusNotification(id, d) {
      return d.get('type', '') == 'request_status'
          && notifKeysExactly(d, ['uid','type','title','message','link','requestId','listingId','status','read','createdAt'])
          && notifBaseOk(d)
          && d.link == 'profile'
          && d.requestId is string && d.requestId != ''
          && d.listingId is string && d.listingId != ''
          && (d.status == 'completed' || d.status == 'rejected')
          && id == 'req_' + d.requestId + '_' + d.status
          && exists(/databases/$(database)/documents/kc_service_requests/$(d.requestId))
          && requestStatusNotifLinksOk(d.uid, d.listingId, d.requestId, d.status);

    // ================================================================
    // ADMINISTRATION FREE ADVERTISEMENT (Spark-compatible, no backend)
    // ----------------------------------------------------------------
    // Two collections:
    //   kc_admin_free_ads/{id}     full campaign records. ADMIN-ONLY
    //                              read/write (drafts, inactive, expired).
    //   kc_adfree_public/current   ONE fixed document = (a) the public
    //                              mirror the dashboard reads and (b) the
    //                              single campaign SLOT.
    //
    // SINGLE-CAMPAIGN LIMIT: there is exactly one path (`current`), so at
    // most one campaign can be live in it. It may be claimed or
    // overwritten by a different campaign only if the current occupant is
    // not published, or its endAt has passed (request.time). Rules are
    // evaluated at commit time against the committed state, so two
    // concurrent publish batches cannot both win.
    // Campaign doc and slot doc are always written in ONE batch and each
    // rule cross-checks the other with getAfter(), so they can never
    // disagree about who holds the slot.
    //
    // TIME WINDOW: a public GET of `current` is allowed only while
    //   published == true && startAt <= request.time < endAt
    // request.time is Firestore SERVER time, so expiry and scheduled start
    // are enforced on every read without any cron/Cloud Function, and a
    // wrong device clock cannot reveal a scheduled ad early. This works
    // because it is a single-document get(), not a list query.
    //
    // Stored status: draft | published | inactive. "scheduled", "active"
    // and "expired" are derived from startAt/endAt + request.time (no
    // backend exists to flip a stored status when time passes).
    //
    // HONEST LIMITATIONS (same family as isValidPhotoUrl above): Rules
    // cannot verify that a Cloudinary-looking URL was really uploaded
    // through the app; and the Cloudinary unsigned preset's own
    // restrictions (formats, size) are a console setting, not code.
    // ================================================================

    function afImageUrlOk(u) {
      return u is string
        && u.size() <= 300
        && u.matches('^https://res\\.cloudinary\\.com/fzntl3lv/image/upload/[A-Za-z0-9/_.-]+\\.(jpg|jpeg|png)$');
    }

    // Optional destination link: https only (stricter than the paid-ad http(s) rule).
    function afOptHttpsUrl(d, f) {
      return !(f in d)
        || (d[f] is string && d[f].size() <= 300 && d[f].matches('^https://[^\\s]+$'));
    }

    function afCampaignKeys() {
      return ['title','description','imageUrl','imageAlt','destinationUrl','status',
              'startAt','endAt','createdBy','createdAt','updatedAt','publishedAt'];
    }

    function afCampaignContentOk(d) {
      return d.keys().hasOnly(afCampaignKeys())
        && d.keys().hasAll(['title','imageUrl','status','startAt','endAt','createdBy','createdAt','updatedAt'])
        && reqStrRange(d.title, 1, 120)
        && optStrMax(d, 'description', 300)
        && optStrMax(d, 'imageAlt', 150)
        && afImageUrlOk(d.imageUrl)
        && afOptHttpsUrl(d, 'destinationUrl')
        && d.createdBy is string
        && d.status in ['draft', 'published', 'inactive']
        && d.startAt is timestamp
        && d.endAt is timestamp
        && d.endAt > d.startAt
        && d.endAt <= d.startAt + duration.value(366, 'd')
        && (!('publishedAt' in d) || d.publishedAt is timestamp);
    }

    function afSlotAfter() {
      return getAfter(/databases/$(database)/documents/kc_adfree_public/current).data;
    }

    function afCampaignAfter(id) {
      return getAfter(/databases/$(database)/documents/kc_admin_free_ads/$(id)).data;
    }

    // Public-visible content must be byte-equal between campaign and slot.
    function afSameContent(a, b) {
      return a.title == b.title
        && a.imageUrl == b.imageUrl
        && a.startAt == b.startAt
        && a.endAt == b.endAt
        && a.get('imageAlt', '') == b.get('imageAlt', '')
        && a.get('description', '') == b.get('description', '')
        && a.get('destinationUrl', '') == b.get('destinationUrl', '');
    }

    function afTransitionOk(o, n) {
      return (o.status == 'draft' && n.status in ['draft', 'published'])
        || (o.status == 'published' && n.status in ['published', 'inactive'])
        || (o.status == 'inactive' && n.status == 'draft');
    }

    function afPublishedAtOk(o, n) {
      return (o.status != 'published' && n.status == 'published' && n.publishedAt == request.time)
        || (!(o.status != 'published' && n.status == 'published')
            && n.get('publishedAt', null) == o.get('publishedAt', null));
    }

    // A campaign that is (or becomes) 'published' must be in the slot, in the same batch.
    function afPublishedWriteOk(id, n) {
      return n.status != 'published'
        || (n.endAt > request.time
            && afSlotAfter().campaignId == id
            && afSlotAfter().published == true
            && afSameContent(afSlotAfter(), n));
    }

    // Leaving 'published' must also release the slot if this campaign still holds it.
    function afLeavingLiveOk(o, id, n) {
      return !(o.status == 'published' && n.status != 'published')
        || afSlotAfter().campaignId != id
        || afSlotAfter().published == false;
    }

    match /kc_admin_free_ads/{id} {

      allow read: if isAdmin();

      allow create: if isAdmin()
                    && afCampaignContentOk(request.resource.data)
                    && request.resource.data.createdBy == request.auth.uid
                    && request.resource.data.createdAt == request.time
                    && request.resource.data.updatedAt == request.time
                    && request.resource.data.status in ['draft', 'published']
                    && (request.resource.data.status == 'draft'
                        || request.resource.data.publishedAt == request.time)
                    && (request.resource.data.status == 'published'
                        || !('publishedAt' in request.resource.data))
                    && afPublishedWriteOk(id, request.resource.data);

      allow update: if isAdmin()
                    && afCampaignContentOk(request.resource.data)
                    && request.resource.data.createdBy == resource.data.createdBy
                    && request.resource.data.createdAt == resource.data.createdAt
                    && request.resource.data.updatedAt == request.time
                    && afTransitionOk(resource.data, request.resource.data)
                    && afPublishedAtOk(resource.data, request.resource.data)
                    && afPublishedWriteOk(id, request.resource.data)
                    && afLeavingLiveOk(resource.data, id, request.resource.data);

      // A live (published, not yet expired) campaign cannot be deleted —
      // deactivate it first, so the slot is always released consistently.
      allow delete: if isAdmin()
                    && (resource.data.status != 'published'
                        || resource.data.endAt <= request.time);
    }

    function afSlotPublicShapeOk(n) {
      return n.keys().hasOnly(['campaignId','published','title','description','imageUrl',
                               'imageAlt','destinationUrl','startAt','endAt','updatedAt'])
        && n.keys().hasAll(['campaignId','published','title','imageUrl','startAt','endAt','updatedAt'])
        && reqStrRange(n.title, 1, 120)
        && optStrMax(n, 'description', 300)
        && optStrMax(n, 'imageAlt', 150)
        && afImageUrlOk(n.imageUrl)
        && afOptHttpsUrl(n, 'destinationUrl')
        && n.startAt is timestamp
        && n.endAt is timestamp
        && n.endAt > n.startAt
        && n.endAt > request.time
        && n.endAt <= n.startAt + duration.value(366, 'd');
    }

    function afSlotWriteOk(n) {
      return n.keys().hasAll(['campaignId', 'published', 'updatedAt'])
        && n.updatedAt == request.time
        && n.campaignId is string
        && n.campaignId.size() > 0
        && n.campaignId.size() <= 100
        && n.published is bool
        && (
          (n.published == true
            && afSlotPublicShapeOk(n)
            && afCampaignAfter(n.campaignId).status == 'published'
            && afSameContent(afCampaignAfter(n.campaignId), n))
          ||
          (n.published == false
            && n.keys().hasOnly(['campaignId', 'published', 'updatedAt'])
            && afCampaignAfter(n.campaignId).status != 'published')
        );
    }

    // The slot may be (re)written by campaign X only if nobody else is live in it.
    function afSlotAvailableFor(old, id) {
      return old.get('published', false) != true
        || old.endAt <= request.time
        || old.campaignId == id;
    }

    match /kc_adfree_public/{docId} {

      allow read: if (
                      docId == 'current'
                      && resource != null
                      && resource.data.get('published', false) == true
                      && resource.data.startAt <= request.time
                      && resource.data.endAt > request.time
                    )
                  || isAdmin();

      allow create: if docId == 'current'
                    && isAdmin()
                    && afSlotWriteOk(request.resource.data);

      allow update: if docId == 'current'
                    && isAdmin()
                    && afSlotWriteOk(request.resource.data)
                    && afSlotAvailableFor(resource.data, request.resource.data.campaignId);

      // Never deleted: deactivate (published=false) instead, so the slot
      // history stays consistent with the campaign records.
      allow delete: if false;
    }

    // ================================================================
    // DEFAULT DENY
    // ================================================================

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
